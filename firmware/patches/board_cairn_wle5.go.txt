//go:build cairn_wle5

package machine

import (
	"device/stm32"
	"runtime/interrupt"
)

// Board file for the CairnMesh node on an STM32WLE5CCU6: the E77-400M22S module
// and, with other constants, the own PCB4 module.
//
// Only what the machine package needs to compile: the console and the unused
// SPI/I2C defaults. The RF switch and the LEDs are in board/wle5 (E77 or PCB4).
// E77-400M22S (EBYTE user manual V1.6, 3.1): PA2/PA3 = USART2 TX/RX (the
// module's AT port), PB6/PB7 = USART1 TX/RX, PB0 = VDDTCXO (must stay open on
// the active-crystal batches), PC14/PC15 = 32.768 kHz crystal on the module.
// PCB4's console pins are not known yet.
// The radio itself is the on-chip SUBGHZ (SUBGHZSPI, no external pins).

// machine.SPI and machine.I2C need default pins to compile; the node uses
// neither (the radio is SUBGHZSPI, driven by hw/subghz). These are the first
// free alternatives on the module and nothing is wired to them by default.
const (
	SPI0_SCK_PIN = PB3 // SPI1 SCK, AF5 (also LED_TX)
	SPI0_SDO_PIN = PB5 // SPI1 MOSI, AF5 (also LED_LINK)
	SPI0_SDI_PIN = PB4 // SPI1 MISO, AF5 (also LED_RX)

	I2C1_SCL_PIN  = PB8 // I2C1 SCL, AF4
	I2C1_SDA_PIN  = PB7 // I2C1 SDA, AF4 (also USART1 RX)
	I2C1_ALT_FUNC = 4
	I2C0_SCL_PIN  = I2C1_SCL_PIN
	I2C0_SDA_PIN  = I2C1_SDA_PIN
)

// UARTs.
const (
	UART1_TX_PIN = PB6
	UART1_RX_PIN = PB7
	UART2_TX_PIN = PA2
	UART2_RX_PIN = PA3

	// The console: USART2 on the module's AT-port pins.
	UART_TX_PIN = UART2_TX_PIN
	UART_RX_PIN = UART2_RX_PIN
)

var (
	// Console UART: USART2 (AF7).
	UART0  = &_UART0
	_UART0 = UART{
		Buffer:            NewRingBuffer(),
		Bus:               stm32.USART2,
		TxAltFuncSelector: AF7_USART1_2,
		RxAltFuncSelector: AF7_USART1_2,
	}
	DefaultUART = UART0
	UART2       = UART0

	UART1  = &_UART1
	_UART1 = UART{
		Buffer:            NewRingBuffer(),
		Bus:               stm32.USART1,
		TxAltFuncSelector: AF7_USART1_2,
		RxAltFuncSelector: AF7_USART1_2,
	}

	I2C1 = &I2C{
		Bus:             stm32.I2C1,
		AltFuncSelector: I2C1_ALT_FUNC,
	}
	I2C0 = I2C1
)

func init() {
	UART0.Interrupt = interrupt.New(stm32.IRQ_USART2, _UART0.handleInterrupt)
	UART1.Interrupt = interrupt.New(stm32.IRQ_USART1, _UART1.handleInterrupt)
}
